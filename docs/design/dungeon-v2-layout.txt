# Dungeon v2 layout (rev 3) - THE DESCENT
#
# Grid-consistent embedding (every edge is a unit step; the in-game map is
# exact). Spawn sits under the roof-cap on the top row; the crown is the
# east end of the bottom row. All ability gating is geometric - no ability
# door locks anywhere; the only explicit gate is the crown coin door.
#
# Cells (x east, y down):
#   y0:            bv(2)  kwp(3)  ga(4)
#   y1:  gr1(1) owp(2) t1(3)  s1(4)  ob(5)  gb(6)
#   y2:  sb(1)  lcc(2) chl(3) lc1(4) wg(5)  swa(6)
#   y3:  sd1(1) mr(2)  gx(3)  sd2(4) ew1(5) gc(6)
#   y4:  kvc1(1) lca(2) lc2(3) owc(4) kvc3(5) swb(6)
#   y5:                 kvc2(3) asl(4) crown(5)
#
# Acts: A (bare) - roof, spawn column, upper halls; glove one door east of
# spawn. B (wall) - the north-west quarter behind the chimney-lock gate,
# with the boots vault at the roof's west end behind the postern. C (dash) -
# the low vault behind the low-ceiling crawl and the sandglass trapdoor.
# The south-east pocket needs dash to enter (sandglass-b) and the astral
# seal before the crown needs both.

# --- rooms -------------------------------------------------------------
room s1    strata-sort-a          # spawn hub, under the greed roof-cap
room ga    greed-loop-a           # roof cap over spawn
room bv    keep-boots-vault       # boots, top-left roof corner (wall-sealed)
room kwp   keep-west-postern      # roof gate: wall-kick bays seal the vault
room gr1   gable-run-a            # north-west gable (wall quarter)
room owp   one-way-loop-c         # vault under-stair (duplicate grid #1)
room t1    tide-shaft-a           # spawn-side shaft, climbable bare
room ob    keep-observatory       # glove, one door from spawn
room gb    greed-loop-b           # east roof step
room sb    strata-sort-b          # wall-quarter crossroads
room lcc   low-ceiling-arena-c    # wall-quarter crawl junction
room chl   chimney-lock-a         # THE wall gate into the north-west quarter
room lc1   lantern-cross-a        # central crossing (duplicate grid #2)
room wg    keep-wall-gate         # wall showcase on the east gallery loop
room swa   switchback-spine-a     # east gallery pocket
room sd1   sandglass-drop-a       # sealed-bulb sandglass: dash trapdoor to the vault
room mr    keep-meteor-run        # dash curtains on the vault's east exit
room gx    greed-loop-a           # under-hall step (duplicate grid #3)
room sd2   sandglass-drop-b       # open-bulb sandglass: dash trapdoor to the deep
room ew1   eaves-walk-a           # east under-eaves walk
room gc    greed-loop-c           # east wall-hatch step down
room kvc1  keyhole-vault-c        # dash vault (coins), west deep
room lca   low-ceiling-arena-a    # dash crawl into the deep
room lc2   lantern-cross-a        # deep crossing (duplicate grid #2)
room owc   one-way-loop-c         # deep junction over the seal
room kvc3  keyhole-vault-c        # east deep vault (duplicate grid #4)
room swb   switchback-spine-b     # east deep cap
room kvc2  keyhole-vault-c        # seal antechamber (duplicate grid #4)
room asl   keep-astral-seal       # final gauntlet: passable only with BOTH
room crown keep-crown-sanctum     # goal, bottom row east end
